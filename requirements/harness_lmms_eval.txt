# lmms-eval's qwen2_5_vl model module imports `decord` unconditionally, even for
# image-only tasks like MMMU and MathVista, so it is required despite no video
# being involved.
#
# NOTE ON PYTHON VERSION: upstream `decord` (0.6.0) ships no wheel for Python
# 3.12+, so on a 3.13 venv `pip install decord` tries to build from source and
# fails. `eva-decord` is a maintained fork exposing the same `decord` module
# with wider wheel coverage. If neither installs, pin an older interpreter with
# eval.harnesses.lmms_eval.python_bin (e.g. python3.11) and rebuild the venv.

eva-decord         # provides `decord`; use plain `decord` if your Python is <=3.11

# latex2sympy2, pinned to antlr4==4.7.2, while latex2sympy2_extended needs
# antlr4 4.9.3-4.13.2 -- no antlr satisfies both. This picks math_verify's side;
# it breaks the old latex2sympy2, which only mathvision/stare/emma import (not
# MMMU or MathVista). Needed by *_reasoning tasks via
# tasks/_task_utils/reasoning_utils.py.
latex2sympy2_extended[antlr4_13_2]==1.11.0
